def remove_duplicates(df):
    return df.drop_duplicates()

def remove_constant_columns(df):
    nunique = df.nunique()
    const_cols = nunique[nunique == 1].index
    return df.drop(columns=const_cols)

def remove_redundant_columns(df):
    cols_to_drop = [
        'COLE_COD_DANE_ESTABLECIMIENTO', 'COLE_COD_DANE_SEDE',
        'ESTU_COD_RESIDE_DEPTO', 'ESTU_COD_RESIDE_MCPIO',
        'ESTU_DEPTO_RESIDE', 'ESTU_PAIS_RESIDE', 'ESTU_NACIONALIDAD',
        'DESEMP_INGLES'
    ]
    cols_to_drop = [c for c in cols_to_drop if c in df.columns]
    return df.drop(columns=cols_to_drop)

def fix_numeric_columns(df):
    score_cols = ['PUNT_GLOBAL', 'PUNT_MATEMATICAS', 'PUNT_LECTURA_CRITICA',
                  'PUNT_SOCIALES_CIUDADANAS', 'PUNT_C_NATURALES', 'PUNT_INGLES']
    for col in score_cols:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors='coerce')
    return df

def filter_boyaca(df):
    if 'ESTU_DEPTO_PRESENTACION' in df.columns:
        return df[df['ESTU_DEPTO_PRESENTACION'].str.upper() == 'BOYACA'].copy()
    return df

def drop_high_missing(df, threshold=0.5):
    return df.dropna(thresh=len(df)*(1-threshold), axis=1)

def clean_categorical_values(df):
    obj_cols = df.select_dtypes(include="object").columns
    df[obj_cols] = df[obj_cols].apply(lambda x: x.str.strip().str.upper())
    return df

def impute_categorical(df):
    cat_cols = df.select_dtypes(include='object').columns
    df[cat_cols] = df[cat_cols].fillna('SIN_DATO')
    return df

def impute_numeric(df):
    num_cols = df.select_dtypes(include='number').columns
    for col in num_cols:
        df[col] = df[col].fillna(df[col].median())
    return df

def save_clean_data(df, path):
    df.to_csv(path, index=False, encoding='utf-8')
    return df

def plot_tech_divide_by_stratum(df):
    """Brecha digital: acceso a internet dentro de cada estrato"""
    plt.figure(figsize=(12, 6))
    ax = sns.barplot(data=df, x='FAMI_ESTRATOVIVIENDA', y='PUNT_GLOBAL',
                     hue='FAMI_TIENEINTERNET', ci=None, palette='Set2',
                     order=['ESTRATO 1', 'ESTRATO 2', 'ESTRATO 3', 'ESTRATO 4', 'ESTRATO 5', 'SIN ESTRATO'])
    plt.title("📡 El acceso a internet impulsa el puntaje, especialmente en estratos bajos", fontweight='bold')
    plt.xlabel("Estrato socioeconómico")
    plt.ylabel("Puntaje promedio global")
    plt.legend(title="¿Tiene Internet?", loc='upper left')
    for p in ax.patches:
        if p.get_height() > 0:
            ax.annotate(f'{p.get_height():.0f}', (p.get_x() + p.get_width()/2., p.get_height()),
                        ha='center', va='center', xytext=(0, 10), textcoords='offset points', fontsize=9)
    plt.tight_layout()
    plt.savefig("insight_brecha_digital_estrato.png")
    plt.close()

def plot_parents_education_comparison(df):
    """Comparación educación madre vs padre"""
    edu_mother = df[['FAMI_EDUCACIONMADRE', 'PUNT_GLOBAL']].rename(columns={'FAMI_EDUCACIONMADRE': 'NIVEL'})
    edu_father = df[['FAMI_EDUCACIONPADRE', 'PUNT_GLOBAL']].rename(columns={'FAMI_EDUCACIONPADRE': 'NIVEL'})
    edu_mother['PARENTESCO'] = 'Madre'
    edu_father['PARENTESCO'] = 'Padre'
    df_edu = pd.concat([edu_mother, edu_father], ignore_index=True)
    order = ['NINGUNO', 'PRIMARIA INCOMPLETA', 'PRIMARIA COMPLETA',
             'SECUNDARIA (BACHILLERATO) INCOMPLETA', 'SECUNDARIA (BACHILLERATO) COMPLETA',
             'TÉCNICA O TECNOLÓGICA INCOMPLETA', 'TÉCNICA O TECNOLÓGICA COMPLETA',
             'EDUCACIÓN PROFESIONAL INCOMPLETA', 'EDUCACIÓN PROFESIONAL COMPLETA', 'POSTGRADO']
    plt.figure(figsize=(14, 6))
    sns.pointplot(data=df_edu, x='NIVEL', y='PUNT_GLOBAL', hue='PARENTESCO',
                  order=order, dodge=True, markers=['o', 's'], linestyles=['-', '--'], ci=None, palette='Set1')
    plt.xticks(rotation=45, ha='right')
    plt.title("👩‍🏫 El nivel educativo de la madre predice mejor el rendimiento que el del padre", fontweight='bold')
    plt.xlabel("Nivel educativo")
    plt.ylabel("Puntaje global promedio")
    plt.tight_layout()
    plt.savefig("insight_educacion_padres.png")
    plt.close()

def plot_tech_accumulation(df):
    """Efecto acumulado de tener computador e internet"""
    df['acceso_tec'] = 'Solo computador'
    df.loc[(df['FAMI_TIENECOMPUTADOR']=='SI') & (df['FAMI_TIENEINTERNET']=='SI'), 'acceso_tec'] = 'Computador + Internet'
    df.loc[(df['FAMI_TIENECOMPUTADOR']=='NO') & (df['FAMI_TIENEINTERNET']=='NO'), 'acceso_tec'] = 'Ninguno'
    df.loc[(df['FAMI_TIENECOMPUTADOR']=='NO') & (df['FAMI_TIENEINTERNET']=='SI'), 'acceso_tec'] = 'Solo internet'
    order = ['Ninguno', 'Solo internet', 'Solo computador', 'Computador + Internet']
    plt.figure(figsize=(12, 6))
    sns.violinplot(data=df, x='acceso_tec', y='PUNT_GLOBAL', order=order,
                   palette='viridis', cut=0, inner='quartile')
    plt.title("📉 La acumulación de recursos tecnológicos multiplica el rendimiento", fontweight='bold')
    plt.xlabel("Acceso en el hogar")
    plt.ylabel("Distribución del puntaje global")
    plt.tight_layout()
    plt.savefig("insight_acumulacion_tecnologica.png")
    plt.close()

def plot_correlation_heatmap(df):
    """Mapa de calor de correlaciones"""
    numeric_cols = ['PUNT_GLOBAL', 'PUNT_MATEMATICAS', 'PUNT_LECTURA_CRITICA',
                    'PUNT_SOCIALES_CIUDADANAS', 'PUNT_C_NATURALES', 'PUNT_INGLES']
    binary_vars = ['FAMI_TIENEINTERNET', 'FAMI_TIENECOMPUTADOR', 'FAMI_TIENELAVADORA', 'FAMI_TIENEAUTOMOVIL']
    for var in binary_vars:
        if var in df.columns:
            df[var + '_BIN'] = (df[var] == 'SI').astype(int)
            numeric_cols.append(var + '_BIN')
    corr = df[numeric_cols].corr()
    plt.figure(figsize=(10, 8))
    mask = np.triu(np.ones_like(corr, dtype=bool))
    sns.heatmap(corr, mask=mask, annot=True, fmt='.2f', cmap='coolwarm',
                square=True, linewidths=.5, cbar_kws={"shrink": 0.8})
    plt.title("🔍 Internet y computador se correlacionan fuertemente con el puntaje global", fontweight='bold')
    plt.tight_layout()
    plt.savefig("insight_correlaciones.png")
    plt.close()

    axes[0].set_ylabel("")

    axes[1].set_title("⚠️ Municipios con mayor rezago educativo", fontweight='bold')

def plot_hacinamiento_impact(df):
    """Impacto del número de personas en el hogar"""
    # Crear categorías de tamaño del hogar y forzar a numérico
    df['tamano_hogar'] = df['FAMI_PERSONASHOGAR'].replace({
        'Uno': 1, 'Dos': 2, 'Tres': 3, 'Cuatro': 4, 'Cinco': 5,
        'Seis': 6, 'Siete': 7, 'Ocho': 8, 'Nueve': 9, 'Diez': 10,
        'Diez o más': 11, 'Doce o más': 12, '9 o más': 9
    })
    
    # Asegurar que ambas columnas sean de tipo numérico
    df['tamano_hogar'] = pd.to_numeric(df['tamano_hogar'], errors='coerce')
    df['PUNT_GLOBAL'] = pd.to_numeric(df['PUNT_GLOBAL'], errors='coerce')
    
    # Eliminar posibles nulos en estas columnas para el gráfico
    df_subset = df.dropna(subset=['tamano_hogar', 'PUNT_GLOBAL'])
    
    # Agrupar por tamaño y calcular puntaje medio
    grouped = df_subset.groupby('tamano_hogar')['PUNT_GLOBAL'].mean().reset_index()
    
    plt.figure(figsize=(12, 6))
    sns.regplot(data=grouped, x='tamano_hogar', y='PUNT_GLOBAL', lowess=True,
                scatter_kws={'s': 80, 'color': '#2c7fb8'}, line_kws={'color': '#d7301f', 'lw': 3})
    plt.title("🏠 A mayor número de personas en el hogar, menor rendimiento académico", fontweight='bold')
    plt.xlabel("Número de miembros en el hogar")
    plt.ylabel("Puntaje global promedio")
    plt.grid(True, linestyle='--', alpha=0.6)
    plt.tight_layout()
    plt.savefig("insight_hacinamiento.png")
    plt.close()

# 4. PIPELINE PRINCIPAL
# ==========================================

def process_pipeline(data_path):
    df = (pd.read_csv(data_path, encoding='utf-8', low_memory=False)
          .pipe(remove_duplicates)
          .pipe(remove_constant_columns)
          .pipe(remove_redundant_columns)
          .pipe(fix_numeric_columns)
          .pipe(filter_boyaca)
          .pipe(drop_high_missing, threshold=0.5)
          .pipe(clean_categorical_values)
          .pipe(impute_categorical)
          .pipe(impute_numeric)
          .pipe(save_clean_data, OUTPUT_DATA)
          )
    # Generar todas las visualizaciones
    plot_tech_divide_by_stratum(df)
    plot_parents_education_comparison(df)
    plot_tech_accumulation(df)
    plot_correlation_heatmap(df)
    plot_municipios_extremos(df)
    plot_colegio_naturaleza_by_internet(df)
    plot_hacinamiento_impact(df)
    plot_boyaca_interactive_map(df)
    return df
